   494→        function fmt(n, decimals = 2) {
   495→            if (n === null || n === undefined || isNaN(n)) return '-';
   496→            if (!isFinite(n)) return n > 0 ? '∞' : '-∞';
   497→            return parseFloat(n).toFixed(decimals);
   498→        }
   499→
   500→        function fmtPrice(n) {
   501→            if (n === null || n === undefined || isNaN(n)) return '-';
   502→            const abs = Math.abs(n);
   503→            const decimals = abs >= 100 ? 2 : abs >= 1 ? 4 : 6;
   504→            return parseFloat(n).toFixed(decimals);
   505→        }

   507→        function fmtDate(d) {
   508→            if (!d) return '-';
   509→            const dt = d instanceof Date ? d : new Date(d);
   510→            return dt.toLocaleString('fa-IR', {
   511→                year: 'numeric', month: '2-digit', day: '2-digit',
   512→                hour: '2-digit', minute: '2-digit'
   513→            });
   514→        }

   857→        function exportCSV() {
   858→            if (!lastReport || !lastReport.trades || lastReport.trades.length === 0) {
   859→                showToast('معامله‌ای برای خروجی وجود ندارد', 'error');
   860→                return;
   861→            }
   862→            const headers = [
   863→                'timestamp', 'symbol', 'direction',
   864→                'entryPrice', 'exitPrice', 'pnl', 'pnlPercent',
   865→                'exitReason', 'capitalAfter'
   866→            ];
   867→            const rows = lastReport.trades.map(t => [
   868→                new Date(t.timestamp).toISOString(),
   869→                t.symbol,
   870→                t.direction,
   871→                t.entryPrice,
   872→                t.exitPrice,
   873→                t.pnl.toFixed(4),
   874→                t.pnlPercent.toFixed(4),
   875→                t.exitReason,
   876→                t.capitalAfter.toFixed(4)
   877→            ]);
   878→            const csv = [headers.join(','), ...rows.map(r => r.join(','))].join('\n');
   879→            const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   880→            const url = URL.createObjectURL(blob);
   881→            const a = document.createElement('a');
   882→            a.href = url;
   883→            a.download = `backtest_${lastReport.config.symbol}_${new Date().toISOString().slice(0, 10)}.csv`;
   884→            document.body.appendChild(a);
   885→            a.click();
   886→            document.body.removeChild(a);
   887→            URL.revokeObjectURL(url);
   888→            showToast('CSV خروجی گرفته شد', 'success');
   889→        }
