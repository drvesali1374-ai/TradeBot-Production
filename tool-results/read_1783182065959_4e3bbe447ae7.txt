    24→## ۱. محدودیت‌های کلودفلر Workers/Pages
    25→
    26→### ۱.۱ محدودیت‌های محیط اجرایی
    27→
    28→| محدودیت | توضیح | تأثیر روی پروژه |
    29→|---------|-------|-----------------|
    30→| **بدون فایل‌سیستم** | `fs` ماژول در Workers وجود ندارد | 🔴 `database.js` کاملاً بی‌استفاده می‌شود |
    31→| **بدون `crypto` ماژول Node.js** | فقط Web Crypto API پشتیبانی می‌شود | 🟠 `helpers.ts` و `history/route.ts` باید بازنویسی شوند |
    32→| **زمان CPU محدود** | 10ms (رایگان) / 30ms (Bundled) | 🟢 API Routes فعلی سریع هستند |
    33→| **زمان دیواری** | 30s (Bundled) / 15min (Unbound) | 🟢 هر درخواست < 5 ثانیه |
    34→| **حجم Worker** | حداکثر 10MB فشرده | 🟢 پروژه فعلی کوچک است |
    35→| **بدون `setTimeout` طولانی** | حداکثر چند ثانیه | 🟢 فقط تأخیرهای 200-300ms در سفارش‌ها |
    36→| **بدون `setInterval` دائمی** | Worker پس از پاسخ خاتمه می‌یابد | 🔵 اتوماسیون باید به Cron منتقل شود |

    39→

    77→```
    78→KV Namespace: TRADING_DATA

    91→### ۲.۳ فایل جدید: `src/lib/tradebot/kv-database.ts`
    92→
    93→```typescript
    94→// ⚠️ این فایل باید جایگزین database.js شود
    95→// اما تا زمان استقرار، هر دو فایل باید وجود داشته باشند
    96→
    97→interface Env {
    98→  TRADING_DATA: KVNamespace
    99→}
   100→
   101→// الگوی KV Repository
   102→export class KVDatabase {
   103→  private kv: KVNamespace
   104→
   105→  constructor(kv: KVNamespace) {
   106→    this.kv = kv
   107→  }
   108→
   109→  // ===== Settings =====
   110→  async getSetting(key: string): Promise<string | null> {
   111→    return this.kv.get(`setting:${key}`)
   112→  }
   113→
   114→  async setSetting(key: string, value: string): Promise<void> {
   115→    await this.kv.put(`setting:${key}`, value)
   116→    // همچنین بروز کل settings کلی
   117→    const all = await this.kv.get('settings') || '{}'
   118→    const parsed = JSON.parse(all)
   119→    parsed[key] = value
   120→    await this.kv.put('settings', JSON.stringify(parsed))
   121→  }
   122→
   123→  // ===== Trades (و سایر آرایه‌ها) =====
   124→  async getTrades(limit = 1000): Promise<any[]> {
   125→    const data = await this.kv.get('trades')
   126→    const trades = data ? JSON.parse(data) : []
   127→    return trades.slice(-limit)
   128→  }
   129→
   130→  async addTrade(trade: any): Promise<{id: number}> {
   131→    const trades = await this.getTrades(Infinity)
   132→    trade.id = trades.length > 0
   133→      ? Math.max(...trades.map(t => t.id || 0)) + 1
   134→      : 1
   135→    trade.created_at = new Date().toISOString()
   136→    trades.push(trade)
   137→    await this.kv.put('trades', JSON.stringify(trades))
   138→    return { id: trade.id }
   139→  }
   140→
   141→  // ... مشابه برای Positions, Signals, Logs, Errors, AutomationState
   142→}
   143→```

   162→### ۲.۵ تابع کارخانه (Factory Pattern) — 🔴 حیاتی
   163→
   164→برای اینکه پروژه هم در محیط محلی (Node.js + فایل JSON) و هم در کلودفلر (KV) کار کند:
   165→
   166→```typescript
   167→// src/lib/tradebot/db-factory.ts
   168→
   169→let dbInstance: DatabaseInterface | null = null
   170→
   171→export interface DatabaseInterface {
   172→  getSetting(key: string): Promise<string | null>
   173→  setSetting(key: string, value: string): Promise<void>
   174→  getTrades(limit?: number): Promise<any[]>
   175→  addTrade(trade: any): Promise<{id: number}>
   176→  // ... سایر متدها
   177→}
   178→
   179→export function getDatabase(kv?: KVNamespace): DatabaseInterface {
   180→  if (!dbInstance) {
   181→    if (kv) {
   182→      // محیط کلودفلر
   183→      dbInstance = new KVDatabase(kv)
   184→    } else {
   185→      // محیط محلی (توسعه)
   186→      dbInstance = new JsonFileDatabase()
   187→    }
   188→  }
   189→  return dbInstance
   190→}
   191→```
   192→
   193→**⚠️ نکته مهم:** این الگو تضمین می‌کند که در توسعه محلی، پروژه بدون تغییر با فایل JSON کار می‌کند و فقط در محیط کلودفلر از KV استفاده می‌شود.
   194→

   209→| محدودیت | راه‌حل |

   211→| **Eventual Consistency** (تأخیر تا ۶۰ ثانیه) | برای writes از `waitUntil()` استفاده شود؛ برای reads از cache حافظه |

   240→## ۳. تغییر ۲: جایگزینی `crypto` Node.js با Web Crypto API
   241→
   242→### ۳.۱ فایل‌های درگیر
   243→
   244→| فایل | استفاده | نوع |
   245→|------|---------|------|
   246→| `src/lib/tradebot/helpers.ts` | `crypto.createHmac('sha256', secretKey)` | 🔴 |
   247→| `src/app/api/history/route.ts` | `crypto.createHmac('sha256', secretKey)` | 🔴 |
   248→
   249→### ۳.۲ تغییر `helpers.ts`
   250→
   251→**کد فعلی:**
   252→```typescript
   253→import crypto from 'crypto'
   254→
   255→export function generateSignature(queryString: string, secretKey: string): string {
   256→  return crypto.createHmac('sha256', secretKey).update(queryString).digest('hex')
   257→}
   258→```
   259→
   260→**کد جدید (سازگار با هر دو محیط):**
   261→```typescript
   262→// تشخیص محیط اجرایی
   263→const isEdgeRuntime = typeof globalThis.crypto?.subtle !== 'undefined'
   264→
   265→export async function generateSignature(queryString: string, secretKey: string): Promise<string> {
   266→  if (isEdgeRuntime) {
   267→    // Web Crypto API (کلودفلر Workers)
   268→    const encoder = new TextEncoder()
   269→    const key = await crypto.subtle.importKey(
   270→      'raw',
   271→      encoder.encode(secretKey),
   272→      { name: 'HMAC', hash: 'SHA-256' },
   273→      false,
   274→      ['sign']
   275→    )
   276→    const signature = await crypto.subtle.sign(
   277→      'HMAC',
   278→      key,
   279→      encoder.encode(queryString)
   280→    )
   281→    return Array.from(new Uint8Array(signature))
   282→      .map(b => b.toString(16).padStart(2, '0'))
   283→      .join('')
   284→  } else {
   285→    // Node.js crypto (توسعه محلی)
   286→    const nodeCrypto = await import('crypto')
   287→    return nodeCrypto.createHmac('sha256', secretKey).update(queryString).digest('hex')
   288→  }
   289→}
   290→```
   291→
   292→### ۳.۳ ⚠️ تغییر مهم: `generateSignature` حالت async دارد
   293→
   294→چون `crypto.subtle.sign` asynchronous است، تمام فراخوانی‌های `generateSignature` باید `await` شوند.
   295→
   296→**فایل‌هایی که باید بروز شوند:**
   297→
   298→| فایل | تغییر |
   299→|------|-------|
   300→| `create-position/route.ts` | ۳ فراخوانی `generateSignature` → `await generateSignature()` |
   301→| `close-position/route.ts` | ۳ فراخوانی `generateSignature` → `await generateSignature()` |
   302→| `balance/route.ts` | ۱ فراخوانی → `await generateSignature()` |
   303→| `open-positions/route.ts` | ۱ فراخوانی → `await generateSignature()` |
   304→

   372→## ۵. تغییر ۴: ایجاد مسیر `/api/db/automation-state`
   373→
   374→### ۵.۱ مشکل
   375→
   376→`automation-manager.js` فراخوانی‌های `dbGet()`, `dbSet()`, `dbDelete()` به `/api/db/automation-state` انجام می‌دهد، اما این مسیر در سرور وجود ندارد و خطای 404 برمی‌گرداند.
   377→
   378→### ۵.۲ فایل جدید: `src/app/api/db/automation-state/route.ts`
   379→
   380→```typescript
   381→import { NextRequest, NextResponse } from 'next/server'
   382→import { AutomationState } from '@/lib/tradebot/database'
   383→
   384→export const runtime = 'edge'  // یا حذف در صورت عدم مهاجرت
   385→
   386→export async function GET(request: NextRequest) {
   387→  const key = request.nextUrl.searchParams.get('key')
   388→  if (!key) {
   389→    const all = AutomationState.getAll()
   390→    return NextResponse.json({ success: true, data: all })
   391→  }
   392→  const value = AutomationState.get(key)
   393→  return NextResponse.json({ success: true, data: value })
   394→}
   395→
   396→export async function POST(request: NextRequest) {
   397→  const { key, value } = await request.json()
   398→  if (!key) return NextResponse.json({ error: 'Key is required' }, { status: 400 })
   399→  AutomationState.set(key, value)
   400→  return NextResponse.json({ success: true })
   401→}
   402→
   403→export async function DELETE(request: NextRequest) {
   404→  const { key } = await request.json()
   405→  if (!key) return NextResponse.json({ error: 'Key is required' }, { status: 400 })
   406→  AutomationState.delete(key)
   407→  return NextResponse.json({ success: true })
   408→}
   409→```
   410→

   412→
   413→`automation-manager.js` پاسخ را اینگونه پارس می‌کند:
   414→```javascript

   416→return data.success ? data.data : null
   417→```
   418→
   419→بنابراین فرمت پاسخ **باید** `{ success: true, data: value }` باشد.

   593→## ۹. تغییر ۸ (آینده): پیاده‌سازی Cron Trigger
   594→
   595→### ۹.۱ توضیح
   596→
   597→اتوماسیون فعلی با `setInterval` در مرورگر کار می‌کند. در کلودفلر، این روش ممکن نیست.
   598→باید از Cron Trigger + Cloudflare Queue استفاده شود (نیاز به پلن پولی $5/ماه).
   599→
   600→### ۹.۲ معماری پیشنهادی
   601→
   602→```
   603→Cron Trigger (5 * * * * UTC)
   604→  → Producer Worker: خواندن نمادها از KV → ارسال به Queue
   605→    → Queue Consumer: پردازش هر نماد (یک چرخه کامل)
   606→      → retry خودکار در صورت خطا
   607→      → Dead Letter Queue برای بررسی دستی
   608→```
   609→
   610→### ۹.۳ wrangler.toml (اضافه‌شده)
   611→
   612→```toml
   613→# Cron Trigger
   614→[triggers]
   615→crons = ["5 * * * *"]
   616→
   617→# Queue
   618→[[queues.producers]]
   619→queue = "trading-cycle"
   620→binding = "CYCLE_QUEUE"
   621→
   622→[[queues.consumers]]
   623→queue = "trading-cycle"
   624→max_batch_size = 1
   625→max_retries = 3
   626→retry_delay = 30
   627→```
   628→
   629→### ۹.۴ منطق Producer Worker
   630→
   631→```typescript
   632→// workers/cron-producer.ts
   633→export default {
   634→  async scheduled(event, env, ctx) {
   635→    // ۱. خواندن لیست نمادها از KV
   636→    const symbolsData = await env.TRADING_DATA.get('automation_symbols')
   637→    const symbols = JSON.parse(symbolsData || '[]')
   638→
   639→    // ۲. ارسال هر نماد به Queue
   640→    for (const symbol of symbols) {
   641→      await env.CYCLE_QUEUE.send({
   642→        symbol: symbol.name,
   643→        timestamp: Date.now()
   644→      })
   645→    }
   646→  }
   647→}
   648→```
   649→
   650→### ۹.۵ منطق Consumer Worker
   651→
   652→```typescript
   653→// workers/cycle-consumer.ts
   654→export default {
   655→  async queue(batch, env) {
   656→    for (const message of batch.messages) {
   657→      const { symbol, timestamp } = message.body
   658→      try {
   659→        await runCycleForSymbol(symbol, env)
   660→        message.ack()
   661→      } catch (error) {
   662→        message.retry()
   663→      }
   664→    }
   665→  }
   666→}
   667→```
   668→
   669→### ۹.۶ ⚠️ نکته مهم: مهاجرت منطق client-side به server-side
   670→
   671→تمام منطق `AutomationManager` (3000+ خط JavaScript) در حال حاضر در مرورگر اجرا می‌شود.
   672→برای Cron Trigger، این منطق باید به TypeScript سمت سرور (Worker) تبدیل شود.
   673→
   674→**اقدامات لازم:**
   675→1. تولید سیگنال → `analyzeMarketData()` → TypeScript Worker
   676→2. کنترل‌های ریسک → TypeScript Worker
   677→3. باز/بسته پوزیشن → API Routes موجود (fetch از Worker)
   678→4. نوتیفیکیشن بله → API Route موجود (fetch از Worker)
   679→5. مدیریت وضعیت نمادها → KV
   680→
   681→**⚠️ این بزرگ‌ترین تغییر پروژه است و باید با دقت انجام شود.**
