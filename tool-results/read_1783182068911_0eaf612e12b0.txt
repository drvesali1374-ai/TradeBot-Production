    26→    26→### ۱.۱ محدودیت‌های محیط اجرایی
    27→    27→
    28→    28→| محدودیت | توضیح | تأثیر روی پروژه |
    29→    29→|---------|-------|-----------------|
    30→    30→| **بدون فایل‌سیستم** | `fs` ماژول در Workers وجود ندارد | 🔴 `database.js` کاملاً بی‌استفاده می‌شود |
    31→    31→| **بدون `crypto` ماژول Node.js** | فقط Web Crypto API پشتیبانی می‌شود | 🟠 `helpers.ts` و `history/route.ts` باید بازنویسی شوند |
    32→    32→| **زمان CPU محدود** | 10ms (رایگان) / 30ms (Bundled) | 🟢 API Routes فعلی سریع هستند |
    33→    33→| **زمان دیواری** | 30s (Bundled) / 15min (Unbound) | 🟢 هر درخواست < 5 ثانیه |
    34→    34→| **حجم Worker** | حداکثر 10MB فشرده | 🟢 پروژه فعلی کوچک است |

    73→    73→### ۲.۲ طراحی جدید: KV + الگوی Repository
    74→    74→
    75→    75→**استراتژی:** هر مجموعه داده در KV با کلید متمایز ذخیره می‌شود.
    76→    76→
    77→    77→```
    78→    78→KV Namespace: TRADING_DATA
    79→    79→├── settings              → JSON (کل تنظیمات)
    80→    80→├── trades                → JSON Array
    81→    81→├── positions             → JSON Array
    82→    82→├── signals               → JSON Array
    83→    83→├── logs                  → JSON Array (آخرین 200 مورد)
    84→    84→├── errors                → JSON Array
    85→    85→├── automationState       → JSON Object (کلید/مقدار)
    86→    86→├── settings:{key}        → مقدار تکی (دسترسی سریع)
    87→    87→├── automationState:{key} → مقدار تکی (دسترسی سریع)
    88→    88→└── stats                 → JSON (کش آمار)
    89→    89→```

    91→    91→### ۲.۳ فایل جدید: `src/lib/tradebot/kv-database.ts`
    92→    92→
    93→    93→```typescript
    94→    94→// ⚠️ این فایل باید جایگزین database.js شود
    95→    95→// اما تا زمان استقرار، هر دو فایل باید وجود داشته باشند
    96→    96→
    97→    97→interface Env {
    98→    98→  TRADING_DATA: KVNamespace
    99→    99→}
   100→   100→
   101→   101→// الگوی KV Repository
   102→   102→export class KVDatabase {
   103→   103→  private kv: KVNamespace
   104→   104→
   105→   105→  constructor(kv: KVNamespace) {
   106→   106→    this.kv = kv
   107→   107→  }
   108→   108→
   109→   109→  // ===== Settings =====
   110→   110→  async getSetting(key: string): Promise<string | null> {
   111→   111→    return this.kv.get(`setting:${key}`)
   112→   112→  }
   113→   113→
   114→   114→  async setSetting(key: string, value: string): Promise<void> {
   115→   115→    await this.kv.put(`setting:${key}`, value)
   116→   116→    // همچنین بروز کل settings کلی
   117→   117→    const all = await this.kv.get('settings') || '{}'
   118→   118→    const parsed = JSON.parse(all)
   119→   119→    parsed[key] = value
   120→   120→    await this.kv.put('settings', JSON.stringify(parsed))
   121→   121→  }
   122→   122→
   123→   123→  // ===== Trades (و سایر آرایه‌ها) =====
   124→   124→  async getTrades(limit = 1000): Promise<any[]> {
   125→   125→    const data = await this.kv.get('trades')
   126→   126→    const trades = data ? JSON.parse(data) : []
   127→   127→    return trades.slice(-limit)
   128→   128→  }
   129→   129→
   130→   130→  async addTrade(trade: any): Promise<{id: number}> {
   131→   131→    const trades = await this.getTrades(Infinity)
   132→   132→    trade.id = trades.length > 0
   133→   133→      ? Math.max(...trades.map(t => t.id || 0)) + 1
   134→   134→      : 1
   135→   135→    trade.created_at = new Date().toISOString()
   136→   136→    trades.push(trade)
   137→   137→    await this.kv.put('trades', JSON.stringify(trades))
   138→   138→    return { id: trade.id }
   139→   139→  }
   140→   140→
   141→   141→  // ... مشابه برای Positions, Signals, Logs, Errors, AutomationState
   142→   142→}
   143→   143→```
   144→   144→
   145→   145→### ۲.۴ نحوه دسترسی به KV در API Routes
   146→   146→
   147→   147→در کلودفلر Workers، KV از طریق `context.env` در دسترس است:
   148→   148→
   149→   149→```typescript
   150→   150→// app/api/db/trades/route.ts
   151→   151→export const runtime = 'edge'
   152→   152→

   154→   154→  // در کلودفلر: KV از طریق bindings در دسترس است
   155→   155→  // در توسعه محلی: از فایل JSON استفاده می‌شود
   156→   156→  const db = getDatabase() // ← تابع کارخانه (factory)

   162→   162→### ۲.۵ تابع کارخانه (Factory Pattern) — 🔴 حیاتی
   163→   163→
   164→   164→برای اینکه پروژه هم در محیط محلی (Node.js + فایل JSON) و هم در کلودفلر (KV) کار کند:
   165→   165→
   166→   166→```typescript
   167→   167→// src/lib/tradebot/db-factory.ts
   168→   168→
   169→   169→let dbInstance: DatabaseInterface | null = null
   170→   170→
   171→   171→export interface DatabaseInterface {
   172→   172→  getSetting(key: string): Promise<string | null>
   173→   173→  setSetting(key: string, value: string): Promise<void>
   174→   174→  getTrades(limit?: number): Promise<any[]>
   175→   175→  addTrade(trade: any): Promise<{id: number}>
   176→   176→  // ... سایر متدها
   177→   177→}
   178→   178→
   179→   179→export function getDatabase(kv?: KVNamespace): DatabaseInterface {
   180→   180→  if (!dbInstance) {
   181→   181→    if (kv) {
   182→   182→      // محیط کلودفلر
   183→   183→      dbInstance = new KVDatabase(kv)
   184→   184→    } else {
   185→   185→      // محیط محلی (توسعه)
   186→   186→      dbInstance = new JsonFileDatabase()
   187→   187→    }
   188→   188→  }
   189→   189→  return dbInstance
   190→   190→}
   191→   191→```
   192→   192→
   193→   193→**⚠️ نکته مهم:** این الگو تضمین می‌کند که در توسعه محلی، پروژه بدون تغییر با فایل JSON کار می‌کند و فقط در محیط کلودفلر از KV استفاده می‌شود.

   240→   240→## ۳. تغییر ۲: جایگزینی `crypto` Node.js با Web Crypto API
   241→   241→
   242→   242→### ۳.۱ فایل‌های درگیر
   243→   243→
   244→   244→| فایل | استفاده | نوع |
   245→   245→|------|---------|------|
   246→   246→| `src/lib/tradebot/helpers.ts` | `crypto.createHmac('sha256', secretKey)` | 🔴 |
   247→   247→| `src/app/api/history/route.ts` | `crypto.createHmac('sha256', secretKey)` | 🔴 |
   248→   248→
   249→   249→### ۳.۲ تغییر `helpers.ts`
   250→   250→
   251→   251→**کد فعلی:**
   252→   252→```typescript
   253→   253→import crypto from 'crypto'
   254→   254→
   255→   255→export function generateSignature(queryString: string, secretKey: string): string {
   256→   256→  return crypto.createHmac('sha256', secretKey).update(queryString).digest('hex')
   257→   257→}
   258→   258→```
   259→   259→
   260→   260→**کد جدید (سازگار با هر دو محیط):**
   261→   261→```typescript
   262→   262→// تشخیص محیط اجرایی
   263→   263→const isEdgeRuntime = typeof globalThis.crypto?.subtle !== 'undefined'
   264→   264→
   265→   265→export async function generateSignature(queryString: string, secretKey: string): Promise<string> {
   266→   266→  if (isEdgeRuntime) {
   267→   267→    // Web Crypto API (کلودفلر Workers)
   268→   268→    const encoder = new TextEncoder()
   269→   269→    const key = await crypto.subtle.importKey(
   270→   270→      'raw',
   271→   271→      encoder.encode(secretKey),
   272→   272→      { name: 'HMAC', hash: 'SHA-256' },
   273→   273→      false,
   274→   274→      ['sign']
   275→   275→    )
   276→   276→    const signature = await crypto.subtle.sign(
   277→   277→      'HMAC',
   278→   278→      key,
   279→   279→      encoder.encode(queryString)
   280→   280→    )
   281→   281→    return Array.from(new Uint8Array(signature))
   282→   282→      .map(b => b.toString(16).padStart(2, '0'))
   283→   283→      .join('')
   284→   284→  } else {
   285→   285→    // Node.js crypto (توسعه محلی)
   286→   286→    const nodeCrypto = await import('crypto')
   287→   287→    return nodeCrypto.createHmac('sha256', secretKey).update(queryString).digest('hex')
   288→   288→  }
   289→   289→}
   290→   290→```
   291→   291→
   292→   292→### ۳.۳ ⚠️ تغییر مهم: `generateSignature` حالت async دارد
   293→   293→
   294→   294→چون `crypto.subtle.sign` asynchronous است، تمام فراخوانی‌های `generateSignature` باید `await` شوند.
   295→   295→
   296→   296→**فایل‌هایی که باید بروز شوند:**
   297→   297→
   298→   298→| فایل | تغییر |
   299→   299→|------|-------|
   300→   300→| `create-position/route.ts` | ۳ فراخوانی `generateSignature` → `await generateSignature()` |
   301→   301→| `close-position/route.ts` | ۳ فراخوانی `generateSignature` → `await generateSignature()` |
   302→   302→| `balance/route.ts` | ۱ فراخوانی → `await generateSignature()` |
   303→   303→| `open-positions/route.ts` | ۱ فراخوانی → `await generateSignature()` |
   304→   304→
   305→   305→### ۳.۴ تغییر `history/route.ts`
   306→   306→
   307→   307→این فایل مستقیماً از `crypto` استفاده می‌کند (بدون helpers):
   308→   308→
   309→   309→**کد فعلی:**

   345→   345→### ۴.۲ نکته درباره `setTimeout` در `create-position/route.ts`
   346→   346→
   347→   347→```typescript
   348→   348→// خط 43:
   349→   349→await new Promise(resolve => setTimeout(resolve, 200)) // تأخیر بعد از تنظیم اهرم
   350→   350→
   351→   351→// خط 91:
   352→   352→await new Promise(resolve => setTimeout(resolve, 200)) // تأخیر بعد از ثبت سفارش
   353→   353→```
   354→   354→
   355→   355→**وضعیت در کلودفلر:** `setTimeout` با تأخیر کم (< 1 ثانیه) در Workers مجاز است. اما اگر Worker خیلی زود خاتمه یابد، ممکن است تأخیر اجرا نشود.
   356→   356→
   357→   357→**راه‌حل:** این تأخیرها برای جلوگیری از rate-limit صرافی هستند و باید حفظ شوند. در Workers، چون کل درخواست در یک handler اجرا می‌شود، این تأخیرها مشکلی ایجاد نمی‌کنند.

   373→   373→
   374→   374→### ۵.۱ مشکل
   375→   375→
   376→   376→`automation-manager.js` فراخوانی‌های `dbGet()`, `dbSet()`, `dbDelete()` به `/api/db/automation-state` انجام می‌دهد، اما این مسیر در سرور وجود ندارد و خطای 404 برمی‌گرداند.
   377→   377→
   378→   378→### ۵.۲ فایل جدید: `src/app/api/db/automation-state/route.ts`
   379→   379→
   380→   380→```typescript
   381→   381→import { NextRequest, NextResponse } from 'next/server'
   382→   382→import { AutomationState } from '@/lib/tradebot/database'
   383→   383→
   384→   384→export const runtime = 'edge'  // یا حذف در صورت عدم مهاجرت
   385→   385→
   386→   386→export async function GET(request: NextRequest) {
   387→   387→  const key = request.nextUrl.searchParams.get('key')
   388→   388→  if (!key) {
   389→   389→    const all = AutomationState.getAll()
   390→   390→    return NextResponse.json({ success: true, data: all })
   391→   391→  }
   392→   392→  const value = AutomationState.get(key)
   393→   393→  return NextResponse.json({ success: true, data: value })
   394→   394→}
   395→   395→
   396→   396→export async function POST(request: NextRequest) {
   397→   397→  const { key, value } = await request.json()
   398→   398→  if (!key) return NextResponse.json({ error: 'Key is required' }, { status: 400 })
   399→   399→  AutomationState.set(key, value)
   400→   400→  return NextResponse.json({ success: true })
   401→   401→}
   402→   402→
   403→   403→export async function DELETE(request: NextRequest) {
   404→   404→  const { key } = await request.json()
   405→   405→  if (!key) return NextResponse.json({ error: 'Key is required' }, { status: 400 })
   406→   406→  AutomationState.delete(key)
   407→   407→  return NextResponse.json({ success: true })
   408→   408→}
   409→   409→```

   593→   593→## ۹. تغییر ۸ (آینده): پیاده‌سازی Cron Trigger
   594→   594→
   595→   595→### ۹.۱ توضیح
   596→   596→
   597→   597→اتوماسیون فعلی با `setInterval` در مرورگر کار می‌کند. در کلودفلر، این روش ممکن نیست.
   598→   598→باید از Cron Trigger + Cloudflare Queue استفاده شود (نیاز به پلن پولی $5/ماه).
   599→   599→
   600→   600→### ۹.۲ معماری پیشنهادی
   601→   601→
   602→   602→```
   603→   603→Cron Trigger (5 * * * * UTC)
   604→   604→  → Producer Worker: خواندن نمادها از KV → ارسال به Queue
   605→   605→    → Queue Consumer: پردازش هر نماد (یک چرخه کامل)
   606→   606→      → retry خودکار در صورت خطا
   607→   607→      → Dead Letter Queue برای بررسی دستی
   608→   608→```
   609→   609→
   610→   610→### ۹.۳ wrangler.toml (اضافه‌شده)
   611→   611→
   612→   612→```toml
   613→   613→# Cron Trigger
   614→   614→[triggers]
   615→   615→crons = ["5 * * * *"]
   616→   616→
   617→   617→# Queue
   618→   618→[[queues.producers]]
   619→   619→queue = "trading-cycle"
   620→   620→binding = "CYCLE_QUEUE"
   621→   621→
   622→   622→[[queues.consumers]]
   623→   623→queue = "trading-cycle"
   624→   624→max_batch_size = 1
   625→   625→max_retries = 3
   626→   626→retry_delay = 30
   627→   627→```
   628→   628→
   629→   629→### ۹.۴ منطق Producer Worker
   630→   630→
   631→   631→```typescript
   632→   632→// workers/cron-producer.ts
   633→   633→export default {
   634→   634→  async scheduled(event, env, ctx) {
   635→   635→    // ۱. خواندن لیست نمادها از KV
   636→   636→    const symbolsData = await env.TRADING_DATA.get('automation_symbols')
   637→   637→    const symbols = JSON.parse(symbolsData || '[]')
   638→   638→
   639→   639→    // ۲. ارسال هر نماد به Queue
   640→   640→    for (const symbol of symbols) {
   641→   641→      await env.CYCLE_QUEUE.send({
   642→   642→        symbol: symbol.name,
   643→   643→        timestamp: Date.now()
   644→   644→      })
   645→   645→    }
   646→   646→  }
   647→   647→}
   648→   648→```
   649→   649→
   650→   650→### ۹.۵ منطق Consumer Worker
   651→   651→
   652→   652→```typescript
   653→   653→// workers/cycle-consumer.ts
   654→   654→export default {
   655→   655→  async queue(batch, env) {
   656→   656→    for (const message of batch.messages) {
   657→   657→      const { symbol, timestamp } = message.body
   658→   658→      try {
   659→   659→        await runCycleForSymbol(symbol, env)
   660→   660→        message.ack()
   661→   661→      } catch (error) {
   662→   662→        message.retry()
   663→   663→      }
   664→   664→    }
   665→   665→  }
   666→   666→}
   667→   667→```
   668→   668→
   669→   669→### ۹.۶ ⚠️ نکته مهم: مهاجرت منطق client-side به server-side
   670→   670→
   671→   671→تمام منطق `AutomationManager` (3000+ خط JavaScript) در حال حاضر در مرورگر اجرا می‌شود.
   672→   672→برای Cron Trigger، این منطق باید به TypeScript سمت سرور (Worker) تبدیل شود.
   673→   673→
   674→   674→**اقدامات لازم:**
   675→   675→1. تولید سیگنال → `analyzeMarketData()` → TypeScript Worker
   676→   676→2. کنترل‌های ریسک → TypeScript Worker
   677→   677→3. باز/بسته پوزیشن → API Routes موجود (fetch از Worker)
   678→   678→4. نوتیفیکیشن بله → API Route موجود (fetch از Worker)
   679→   679→5. مدیریت وضعیت نمادها → KV
   680→   680→
   681→   681→**⚠️ این بزرگ‌ترین تغییر پروژه است و باید با دقت انجام شود.**
