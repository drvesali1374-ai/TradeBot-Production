    42→    async init() {
    43→        await this.loadSettings();
    44→        await this.loadSymbols();
    45→        this.loadLogsFromDatabase();
    46→        this.initChart();
    47→        this.bindEvents();
    48→        this.renderSymbolsTable();

    50→
    51→        // Load balance and open positions display (top bar + positions table)
    52→        this.loadBalanceFromDB();
    53→        this.fetchBalanceForBar();
    54→        // First load position history for all symbols (needed for last entry price cache)
    55→        // ✓ loadHistoryForOpenPositions no longer modifies currentSymbolHistory (race condition fixed)
    56→        this.loadHistoryForOpenPositions().then(() => {
    57→            // Then load and display open positions (which uses the history cache)
    58→            this.loadOpenPositionsFromDB();
    59→            this.fetchOpenPositionsForTable();
    60→        });
    61→        
    62→        // ✓ Try to load previous data for the last used symbol
    63→        //   lastUsedSymbol is read from DB first (priority), then localStorage as fallback.
    64→        //   This runs AFTER loadHistoryForOpenPositions to avoid race condition:
    65→        //   loadHistoryForOpenPositions populates lastEntryPriceCache without touching currentSymbolHistory,
    66→        //   then loadAutomationData sets currentSymbolHistory/Signals/marketData for lastUsedSymbol.
    67→        const dbLastUsedSymbol = await this.dbGet('lastUsedSymbol');
    68→        this.lastUsedSymbol = dbLastUsedSymbol || localStorage.getItem('automationLastUsedSymbol');
    69→        
    70→        if (this.lastUsedSymbol) {
    71→            // Try to load data for the last used symbol
    72→            this.loadAutomationData(this.lastUsedSymbol).then(loaded => {
    73→                if (loaded) {
    74→                    this.restoreUIWithLoadedData(this.lastUsedSymbol);
    75→                }
    76→            });
    77→        } else if (this.symbols.length > 0) {
    78→            // Fallback to first symbol if no last used symbol
    79→            const firstSymbol = this.symbols[0];
    80→            this.loadAutomationData(firstSymbol.name).then(loaded => {
    81→                if (loaded) {
    82→                    this.restoreUIWithLoadedData(firstSymbol.name);
    83→                }
    84→            });
    85→        }
    86→
    87→        // ✓ Cloudflare Bale fix: Send pending Bale messages from the browser
    88→        // The server (Cron Trigger) can't reach tapi.bale.ai from Cloudflare,
    89→        // so it stores messages in KV. The browser (in Iran) can reach Bale API,
    90→        // so we pick up pending messages and send them directly.
    91→        this.sendPendingBaleMessages();
    92→    }

    94→    /**
    95→     * ✓ Cloudflare Bale fix: Send pending Bale messages from the browser
    96→     *
    97→     * پیام‌های بله که از سرور (Cron) قابل ارسال نبودند را از KV می‌خواند
    98→     * و مستقیماً از مرورگر به Bale API ارسال می‌کند.
    99→     */
   100→    async sendPendingBaleMessages() {
   101→        try {
   102→            const pending = await this.dbGet('pendingBaleMessages');
   103→            if (!pending || !Array.isArray(pending) || pending.length === 0) return;
   104→
   105→            console.log(`[BaleFix] Found ${pending.length} pending Bale messages to send from browser`);
   106→
   107→            let sentCount = 0;
   108→            const failedMessages = [];
   109→
   110→            for (const msg of pending) {
   111→                try {
   112→                    const baleUrl = `https://tapi.bale.ai/bot${msg.token}/sendMessage`;
   113→                    await fetch(baleUrl, {
   114→                        method: 'POST',
   115→                        mode: 'no-cors', // ✓ bypass CORS
   116→                        headers: { 'Content-Type': 'text/plain' },
   117→                        body: JSON.stringify({
   118→                            chat_id: String(msg.chatId),
   119→                            text: String(msg.text)
   120→                        })
   121→                    });
   122→                    sentCount++;
   123→                } catch (e) {
   124→                    // Network error — keep the message for next attempt
   125→                    failedMessages.push(msg);
   126→                }
   127→            }
   128→
   129→            console.log(`[BaleFix] Sent ${sentCount}/${pending.length} pending messages`);
   130→
   131→            // Clear sent messages, keep only failed ones (for retry next time)
   132→            await this.dbSet('pendingBaleMessages', failedMessages);
   133→
   134→            if (sentCount > 0) {
   135→                this.log(`${sentCount} پیام بله معوق از مرورگر ارسال شد`, 'success');
   136→            }
   137→        } catch (e) {
   138→            console.warn('[BaleFix] Error sending pending Bale messages:', e);
   139→        }
   140→    }

   142→    restoreUIWithLoadedData(symbolName) {
   143→        // ✓ Restore ALL 8 sections of the automation page from loaded data.
   144→        //   This is called after loadAutomationData() successfully loads data
   145→        //   from the permanent database, so navigating away and returning to
   146→        //   the automation page shows the most recent data.
   147→
   148→        let restoredSections = [];
   149→
   150→        // 1. "داده‌های بازار" — market data table
   151→        if (this.currentSymbolData && this.currentSymbolData.length > 0) {
   152→            this.populateMarketDataTable();
   153→            restoredSections.push('داده‌های بازار');
   154→        }
   155→
   156→        // 2. "جزئیات سیگنال‌ها" + 5. "نمودار قیمت و سیگنال‌ها"
   157→        //    (chart is rendered from currentSymbolData + signals)
   158→        if (this.signals && this.signals.length > 0) {
   159→            this.renderSignalDetails();
   160→            this.renderChart();
   161→            this.updateChartSymbolName(symbolName);
   162→            restoredSections.push('جزئیات سیگنال‌ها + نمودار');
   163→        }
   164→
   165→        // 4. "سیگنال آماده برای ورود" — ready signal card
   166→        //    ✓ This is the key addition: previously this section was NOT restored
   167→        //      on page reload, so the ready-signal card was always empty when
   168→        //      returning to the automation page. Now it's restored from DB.
   169→        this.renderSelectedSignal();
   170→        if (this.selectedSignal) {
   171→            restoredSections.push('سیگنال آماده');
   172→        }
   173→
   174→        // 3. "سوابق پوزیشن‌ها" — history table
   175→        if (this.currentSymbolHistory && this.currentSymbolHistory.length > 0) {
   176→            this.populateHistoryTable();
   177→            restoredSections.push('سوابق پوزیشن‌ها');
   178→        }
   179→
   180→        if (restoredSections.length > 0) {
   181→            this.log(`${symbolName}: داده‌های قبلی بارگذاری شدند (${restoredSections.join('، ')})`, 'success', symbolName);
   182→        }
   183→    }

   247→    async loadSettings() {
   248→        const defaults = this.getDefaultSettings();
   249→        // Start with defaults
   250→        this.settings = { ...defaults };
   251→        
   252→        // Override with localStorage if exists
   253→        const saved = localStorage.getItem('automation_settings');
   254→        if (saved) {
   255→            try {
   256→                const parsed = JSON.parse(saved);
   257→                this.settings = { ...defaults, ...parsed };
   258→            } catch (e) { /* ignore */ }
   259→        }
   260→        
   261→        // Override with DB if exists (DB takes priority)
   262→        try {
   263→            const dbSettings = await this.dbGet('automation_settings');
   264→            if (dbSettings) {
   265→                const parsed = typeof dbSettings === 'string' ? JSON.parse(dbSettings) : dbSettings;
   266→                this.settings = { ...defaults, ...parsed };
   267→            }
   268→        } catch (e) { /* ignore */ }
   269→
   270→        // ✓ Cloudflare-ready: Also sync marketSignalSettings from DB (indicator params + API keys).
   271→        // This ensures the automation page uses the same API keys / indicator config that were
   272→        // saved on the settings page, persisted in the permanent database.
   273→        try {
   274→            const dbMarketSettings = await this.dbGet('marketSignalSettings');
   275→            if (dbMarketSettings) {
   276→                const parsed = typeof dbMarketSettings === 'string' ? JSON.parse(dbMarketSettings) : dbMarketSettings;
   277→                const local = JSON.parse(localStorage.getItem('marketSignalSettings') || '{}');
   278→                // DB takes priority; merge over localStorage
   279→                const merged = { ...local, ...parsed };
   280→                localStorage.setItem('marketSignalSettings', JSON.stringify(merged));
   281→            }
   282→        } catch (e) { /* ignore */ }
   283→        
   284→        this.populateSettingsForm();
   285→    }

   429→    addSymbol(name) {
   430→        if (!name || name.trim() === '') {
   431→            this.log('نام نماد نمی‌تواند خالی باشد', 'error');
   432→            return;
   433→        }
   434→
   435→        name = name.trim().toUpperCase();
   436→        
   437→        if (this.symbols.some(s => s.name === name)) {
   438→            this.log(`نماد ${name} قبلاً اضافه شده است`, 'warning');
   439→            return;
   440→        }
   441→
   442→        const newSymbol = {
   443→            id: this.symbols.length > 0 ? Math.max(...this.symbols.map(s => s.id)) + 1 : 1,
   444→            name: name,
   445→            status: 'waiting',
   446→            errorCount: 0,
   447→            lastCycleTime: null
   448→        };
   449→
   450→        this.symbols.push(newSymbol);
   451→        this.saveSymbols();
   452→        this.renderSymbolsTable();
   453→        this.log(`نماد ${name} با موفقیت اضافه شد`, 'success');
   454→        document.getElementById('new-symbol-name').value = '';
   455→    }

   465→        this.renderSymbolsTable();
   466→        this.log('نماد با موفقیت حذف شد', 'success');
   467→    }

   481→        const waitTimeMs = this.settings.tradeWaitTime * 60 * 1000;
   482→
   483→        const readySymbols = this.symbols.filter(symbol => {
   484→            if (!symbol.lastCycleTime) {
   485→                return true;
   486→            }
   487→
   488→            const elapsedTime = now - symbol.lastCycleTime;
   489→            return elapsedTime >= waitTimeMs;
   490→        });
   491→
   492→        return readySymbols.length > 0 ? readySymbols[0] : null;
   493→    }

   495→    getRemainingWaitTime(symbol) {
   496→        if (!symbol.lastCycleTime) return 0;
   497→        
   498→        const now = Date.now();
   499→        const waitTimeMs = this.settings.tradeWaitTime * 60 * 1000;
   500→        const elapsedTime = now - symbol.lastCycleTime;
   501→        const remainingTime = waitTimeMs - elapsedTime;
   502→        
   503→        return Math.max(0, Math.ceil(remainingTime / 60000));
   504→    }

   550→    /**
   551→     * تحلیل داده‌های بازار و تولید تمام سیگنال‌ها
   552→     * (مطابق analyzeData در داشبورد)
   553→     */
   554→    analyzeMarketData(symbol) {
   555→        if (this.marketData.length === 0) {
   556→            throw new Error('داده‌های بازار موجود نیست');
   557→        }
   558→
   559→        const settings = JSON.parse(localStorage.getItem('marketSignalSettings')) || {};
   560→        const N = this.marketData.length;
   561→        
   562→        // تبدیل داده‌ها به فرمت استفاده شده
   563→        const data = this.marketData.map(c => ({
   564→            timestamp: new Date(c[0]),
   565→            open: parseFloat(c[1]),
   566→            high: parseFloat(c[2]),
   567→            low: parseFloat(c[3]),
   568→            close: parseFloat(c[4]),
   569→            amount: parseFloat(c[5])
   570→        }));
   571→
   572→        // Calculate daily data
   573→        const offset = 3.5 * 3600 * 1000;
   574→        const dailyData = {};
   575→        for (let i = 0; i < N; i++) {
   576→            const localTs = data[i].timestamp.getTime() + offset;
   577→            const day = Math.floor(localTs / 86400000);
   578→            if (!dailyData[day]) {
   579→                dailyData[day] = {
   580→                    maxHigh: -Infinity,
   581→                    minLow: Infinity,
   582→                    lastClose: 0,
   583→                    lastTs: -Infinity
   584→                };
   585→            }
   586→            dailyData[day].maxHigh = Math.max(dailyData[day].maxHigh, data[i].high);
   587→            dailyData[day].minLow = Math.min(dailyData[day].minLow, data[i].low);
   588→            if (localTs > dailyData[day].lastTs) {
   589→                dailyData[day].lastTs = localTs;
   590→                dailyData[day].lastClose = data[i].close;
   591→            }
   592→        }
   593→
   594→        const days = Object.keys(dailyData).sort((a, b) => a - b).map(Number);
   595→
   596→        const prevDailyHighs = [];
   597→        const prevDailyLows = [];
   598→        const prevDayCloses = [];
   599→        for (let i = 0; i < N; i++) {
   600→            const localTs = data[i].timestamp.getTime() + offset;
   601→            const day = Math.floor(localTs / 86400000);
   602→            const prevDayIndex = days.indexOf(day) - 1;
   603→            if (prevDayIndex >= 0) {
   604→                const prevDay = days[prevDayIndex];
   605→                prevDailyHighs.push(dailyData[prevDay].maxHigh);
   606→                prevDailyLows.push(dailyData[prevDay].minLow);
   607→                prevDayCloses.push(dailyData[prevDay].lastClose);
   608→            } else {
   609→                prevDailyHighs.push(null);
   610→                prevDailyLows.push(null);
   611→                prevDayCloses.push(null);
   612→            }
   613→        }
   614→
   615→        // Calculate indicators
   616→        const atr = SignalUtils.calculateATR(data, settings.atrPeriod || 14);
   617→        const avgVols = SignalUtils.calculateSMA(data.map(d => d.amount), settings.avgVolPeriod || 50);
   618→        const rsi = SignalUtils.calculateRSI(data, settings.rsiPeriod || 14);
   619→
   620→        // Track crossovers (kept for backward compatibility — used when filters are disabled)
   621→        let lastUL = null, lastOH = null;
   622→        const lastCrossUnderPL = new Array(N).fill(Infinity);
   623→        const lastCrossOverPH = new Array(N).fill(Infinity);
   624→        for (let i = 1; i < N; i++) {
   625→            if (data[i].close < prevDailyLows[i] && data[i - 1].close >= prevDailyLows[i - 1]) lastUL = i;
   626→            if (data[i].close > prevDailyHighs[i] && data[i - 1].close <= prevDailyHighs[i - 1]) lastOH = i;
   627→            lastCrossUnderPL[i] = lastUL === null ? Infinity : i - lastUL;
   628→            lastCrossOverPH[i] = lastOH === null ? Infinity : i - lastOH;
   629→        }
   630→
   631→        // ✓ Fake Breakout settings (read from this.settings — Fake Breakout fields live there)
   632→        const fbSettings = (this && this.settings) ? this.settings : {};
   633→        const enableMeaningfulBreakFilter = fbSettings.enableMeaningfulBreakFilter !== false;
   634→        const breakAtrMultiplier = (fbSettings.breakAtrMultiplier !== undefined && fbSettings.breakAtrMultiplier !== null)
   635→            ? fbSettings.breakAtrMultiplier : 0.20;
   636→        const breakDetectionMethod = fbSettings.breakDetectionMethod === 'Close' ? 'Close' : 'Wick';
   637→        const enableBreakLifecycleManagement = fbSettings.enableBreakLifecycleManagement !== false;
   638→        const breakSequenceLifetime = (fbSettings.breakSequenceLifetime !== undefined && fbSettings.breakSequenceLifetime !== null)
   639→            ? fbSettings.breakSequenceLifetime : 0;
   640→
   641→        // ✓ Track meaningful breaks (used when enableMeaningfulBreakFilter is true)
   642→        //   A meaningful break for Long = price goes BELOW prevDailyLow by (atr * breakAtrMultiplier)
   643→        //   A meaningful break for Short = price goes ABOVE prevDailyHigh by (atr * breakAtrMultiplier)
   644→        const lastMeaningfulBreakLong = new Array(N).fill(Infinity);
   645→        const lastMeaningfulBreakShort = new Array(N).fill(Infinity);
   646→        {
   647→            let lastMBL = null, lastMBS = null;
   648→            for (let i = 1; i < N; i++) {
   649→                const bd = (atr[i] || 0) * breakAtrMultiplier;
   650→                let mbl = false, mbs = false;
   651→                if (breakDetectionMethod === 'Wick') {
   652→                    mbl = prevDailyLows[i] !== null && data[i].low <= (prevDailyLows[i] - bd);
   653→                    mbs = prevDailyHighs[i] !== null && data[i].high >= (prevDailyHighs[i] + bd);
   654→                } else {
   655→                    mbl = prevDailyLows[i] !== null && data[i].close <= (prevDailyLows[i] - bd);
   656→                    mbs = prevDailyHighs[i] !== null && data[i].close >= (prevDailyHighs[i] + bd);
   657→                }
   658→                if (mbl) lastMBL = i;
   659→                if (mbs) lastMBS = i;
   660→                lastMeaningfulBreakLong[i] = lastMBL === null ? Infinity : i - lastMBL;
   661→                lastMeaningfulBreakShort[i] = lastMBS === null ? Infinity : i - lastMBS;
   662→            }
   663→        }
   664→
   665→        // Generate signals
   666→        this.signals = [];
   667→        const htfSource = settings.htfConfirmationSource || 'signalCandleClose';
   668→
   669→        // ✓ Break Lifecycle State Machine (used when enableBreakLifecycleManagement is true)
   670→        const resetBreakState = () => ({
   671→            active: false,
   672→            type: null,           // 'Long' | 'Short' | null
   673→            breakCandleIndex: null,
   674→            waitingRecovery: false,
   675→            recovered: false,
   676→            recoveryCandleIndex: null
   677→        });
   678→        let breakState = resetBreakState();
   679→
   680→        for (let i = 1; i < N - 1; i++) {
   681→            // HTF confirmation (always computed — used by both branches)
   682→            let htfConfirmLong, htfConfirmShort;
   683→            if (htfSource === 'previousDayClose') {
   684→                htfConfirmLong = prevDayCloses[i] !== null && prevDayCloses[i] > prevDailyLows[i];
   685→                htfConfirmShort = prevDayCloses[i] !== null && prevDayCloses[i] < prevDailyHighs[i];
   686→            } else {
   687→                // Default: Signal Candle Close
   688→                htfConfirmLong = prevDailyLows[i] !== null && data[i].close > prevDailyLows[i];
   689→                htfConfirmShort = prevDailyHighs[i] !== null && data[i].close < prevDailyHighs[i];
   690→            }
   691→
   692→            data[i].rsi = rsi[i];
   693→            data[i].atr = atr[i];
   694→
   695→            if (enableBreakLifecycleManagement) {
   696→                // ===========================================================
   697→                // ✓ Fake Breakout: Break Lifecycle Management state machine
   698→                //   Signal is generated at the RECOVERY candle (price closing
   699→                //   back inside the prior day range), not at the break candle.
   700→                // ===========================================================
   701→
   702→                // Compute meaningful break conditions for this candle
   703→                const breakDistance = (atr[i] || 0) * breakAtrMultiplier;
   704→                let isMeaningfulBreakLong = false;
   705→                let isMeaningfulBreakShort = false;
   706→                if (breakDetectionMethod === 'Wick') {
   707→                    isMeaningfulBreakLong = prevDailyLows[i] !== null && data[i].low <= (prevDailyLows[i] - breakDistance);
   708→                    isMeaningfulBreakShort = prevDailyHighs[i] !== null && data[i].high >= (prevDailyHighs[i] + breakDistance);
   709→                } else {
   710→                    isMeaningfulBreakLong = prevDailyLows[i] !== null && data[i].close <= (prevDailyLows[i] - breakDistance);
   711→                    isMeaningfulBreakShort = prevDailyHighs[i] !== null && data[i].close >= (prevDailyHighs[i] + breakDistance);
   712→                }
   713→
   714→                // 1. Check expiry — if a break has been waiting too long without recovery, discard it
   715→                if (breakState.active && !breakState.recovered && breakState.breakCandleIndex !== null) {
   716→                    const lifetime = breakSequenceLifetime > 0 ? breakSequenceLifetime : (settings.lookback || 50);
   717→                    if (i - breakState.breakCandleIndex > lifetime) {
   718→                        breakState = resetBreakState();
   719→                    }
   720→                }
   721→
   722→                // 2. Check for new meaningful break (only if no active break, or previous one recovered/consumed)
   723→                if (!breakState.active || breakState.recovered) {
   724→                    if (isMeaningfulBreakLong) {
   725→                        breakState = {
   726→                            active: true, type: 'Long', breakCandleIndex: i,
   727→                            waitingRecovery: true, recovered: false, recoveryCandleIndex: null
   728→                        };
   729→                    } else if (isMeaningfulBreakShort) {
   730→                        breakState = {
   731→                            active: true, type: 'Short', breakCandleIndex: i,
   732→                            waitingRecovery: true, recovered: false, recoveryCandleIndex: null
   733→                        };
   734→                    }
   735→                }
   736→
   737→                // 3. Check for new break while waiting recovery (replace old break if OPPOSITE direction)
   738→                if (breakState.active && !breakState.recovered) {
   739→                    if (breakState.type === 'Long' && isMeaningfulBreakShort) {
   740→                        breakState = {
   741→                            active: true, type: 'Short', breakCandleIndex: i,
   742→                            waitingRecovery: true, recovered: false, recoveryCandleIndex: null
   743→                        };
   744→                    } else if (breakState.type === 'Short' && isMeaningfulBreakLong) {
   745→                        breakState = {
   746→                            active: true, type: 'Long', breakCandleIndex: i,
   747→                            waitingRecovery: true, recovered: false, recoveryCandleIndex: null
   748→                        };
   749→                    }
   750→                }
   751→
   752→                // 4. Check recovery — price closes back inside the prior day range
   753→                if (breakState.active && breakState.waitingRecovery) {
   754→                    if (breakState.type === 'Long') {
   755→                        // Recovery: price closes back above Previous Daily Low
   756→                        if (data[i].close > prevDailyLows[i]) {
   757→                            breakState.recovered = true;
   758→                            breakState.recoveryCandleIndex = i;
   759→                            breakState.waitingRecovery = false;
   760→                        }
   761→                    } else if (breakState.type === 'Short') {
   762→                        // Recovery: price closes back below Previous Daily High
   763→                        if (data[i].close < prevDailyHighs[i]) {
   764→                            breakState.recovered = true;
   765→                            breakState.recoveryCandleIndex = i;
   766→                            breakState.waitingRecovery = false;
   767→                        }
   768→                    }
   769→                }
   770→
   771→                // 5. Generate signal ONLY if recovered, on the recovery candle
   772→                if (breakState.recovered && breakState.recoveryCandleIndex === i) {
   773→                    if (breakState.type === 'Long') {
   774→                        const condLong = avgVols[i] !== null
   775→                            && data[i].amount > avgVols[i] * (settings.volMult || 0.2)
   776→                            && rsi[i] !== null
   777→                            && rsi[i] < (settings.rsiThreshold || 50)
   778→                            && htfConfirmLong;
   779→
   780→                        if (condLong) {
   781→                            data[i].signal = 'Long';
   782→                            if (settings.longFixedTp !== null && !isNaN(settings.longFixedTp)) {
   783→                                data[i].tp = data[i].close + (data[i].close * (settings.longFixedTp / 100));
   784→                            } else {
   785→                                data[i].tp = data[i].close + (atr[i] * (settings.tpLongMult || 20));
   786→                            }
   787→                            if (settings.longFixedSl !== null && !isNaN(settings.longFixedSl)) {
   788→                                data[i].sl = data[i].close - (data[i].close * (settings.longFixedSl / 100));
   789→                            } else {
   790→                                data[i].sl = data[i].close - (atr[i] * (settings.slLongMult || 6));
   791→                            }
   792→                            data[i].clientOrderId = SignalUtils.generateOrderId(data[i].timestamp, symbol);
   793→                            this.signals.push({
   794→                                type: 'Long',
   795→                                timestamp: data[i].timestamp,
   796→                                price: data[i].close,
   797→                                tp: data[i].tp,
   798→                                sl: data[i].sl,
   799→                                orderId: data[i].clientOrderId,
   800→                                symbol: symbol,
   801→                                // ✓ مورد ۶: candle index — used by min-same-direction-candles risk control
   802→                                candleIndex: i
   803→                            });
   804→                        }
   805→                    } else if (breakState.type === 'Short') {
   806→                        const condShort = avgVols[i] !== null
   807→                            && data[i].amount > avgVols[i] * (settings.volMult || 0.2)
   808→                            && rsi[i] !== null
   809→                            && rsi[i] > (settings.rsiThreshold || 50)
   810→                            && htfConfirmShort;
   811→
   812→                        if (condShort) {
   813→                            data[i].signal = 'Short';
   814→                            if (settings.shortFixedTp !== null && !isNaN(settings.shortFixedTp)) {
   815→                                data[i].tp = data[i].close - (data[i].close * (settings.shortFixedTp / 100));
   816→                            } else {
   817→                                data[i].tp = data[i].close - (atr[i] * (settings.tpShortMult || 24));
   818→                            }
   819→                            if (settings.shortFixedSl !== null && !isNaN(settings.shortFixedSl)) {
   820→                                data[i].sl = data[i].close + (data[i].close * (settings.shortFixedSl / 100));
   821→                            } else {
   822→                                data[i].sl = data[i].close + (atr[i] * (settings.slShortMult || 4));
   823→                            }
   824→                            data[i].clientOrderId = SignalUtils.generateOrderId(data[i].timestamp, symbol);
   825→                            this.signals.push({
   826→                                type: 'Short',
   827→                                timestamp: data[i].timestamp,
   828→                                price: data[i].close,
   829→                                tp: data[i].tp,
   830→                                sl: data[i].sl,
   831→                                orderId: data[i].clientOrderId,
   832→                                symbol: symbol,
   833→                                // ✓ مورد ۶: candle index — used by min-same-direction-candles risk control
   834→                                candleIndex: i
   835→                            });
   836→                        }
   837→                    }
   838→
   839→                    // Reset after signal generation (break is consumed whether or not signal was generated)
   840→                    breakState = resetBreakState();
   841→                }
   842→            } else {
   843→                // ===========================================================
   844→                // ✓ Backward compatible: original logic (with optional meaningful break filter)
   845→                //   When enableMeaningfulBreakFilter is false → EXACT original behavior
   846→                //   When enableMeaningfulBreakFilter is true  → prior-break crossover check
   847→                //     is replaced with the meaningful break check (filter out weak breaks)
   848→                // ===========================================================
   849→                const isCrossOverPL = data[i].close > prevDailyLows[i] && data[i - 1].close <= prevDailyLows[i - 1];
   850→                const isCrossUnderPH = data[i].close < prevDailyHighs[i] && data[i - 1].close >= prevDailyHighs[i - 1];
   851→
   852→                // Recent prior break — use meaningful break tracker when filter is enabled,
   853→                // otherwise use plain crossover tracker (original behavior)
   854→                const recentBreakLong = enableMeaningfulBreakFilter
   855→                    ? lastMeaningfulBreakLong[i]
   856→                    : lastCrossUnderPL[i];
   857→                const recentBreakShort = enableMeaningfulBreakFilter
   858→                    ? lastMeaningfulBreakShort[i]
   859→                    : lastCrossOverPH[i];
   860→
   861→                const condLong = isCrossOverPL
   862→                    && recentBreakLong <= (settings.lookback || 50)
   863→                    && avgVols[i] !== null
   864→                    && data[i].amount > avgVols[i] * (settings.volMult || 0.2)
   865→                    && rsi[i] !== null
   866→                    && rsi[i] < (settings.rsiThreshold || 50)
   867→                    && htfConfirmLong;
   868→
   869→                const condShort = isCrossUnderPH
   870→                    && recentBreakShort <= (settings.lookback || 50)
   871→                    && avgVols[i] !== null
   872→                    && data[i].amount > avgVols[i] * (settings.volMult || 0.2)
   873→                    && rsi[i] !== null
   874→                    && rsi[i] > (settings.rsiThreshold || 50)
   875→                    && htfConfirmShort;
   876→
   877→                if (condLong) {
   878→                    data[i].signal = 'Long';
   879→                    if (settings.longFixedTp !== null && !isNaN(settings.longFixedTp)) {
   880→                        data[i].tp = data[i].close + (data[i].close * (settings.longFixedTp / 100));
   881→                    } else {
   882→                        data[i].tp = data[i].close + (atr[i] * (settings.tpLongMult || 20));
   883→                    }
   884→                    if (settings.longFixedSl !== null && !isNaN(settings.longFixedSl)) {
   885→                        data[i].sl = data[i].close - (data[i].close * (settings.longFixedSl / 100));
   886→                    } else {

   888→                    }
   889→                    data[i].clientOrderId = SignalUtils.generateOrderId(data[i].timestamp, symbol);
   890→                    this.signals.push({
   891→                        type: 'Long',
   892→                        timestamp: data[i].timestamp,
   893→                        price: data[i].close,
   894→                        tp: data[i].tp,
   895→                        sl: data[i].sl,
   896→                        orderId: data[i].clientOrderId,

   898→                        // ✓ مورد ۶: candle index — used by min-same-direction-candles risk control

   900→                    });
   901→                } else if (condShort) {
   902→                    data[i].signal = 'Short';
   903→                    if (settings.shortFixedTp !== null && !isNaN(settings.shortFixedTp)) {
   904→                        data[i].tp = data[i].close - (data[i].close * (settings.shortFixedTp / 100));

   910→                    } else {
   911→                        data[i].sl = data[i].close + (atr[i] * (settings.slShortMult || 4));
   912→                    }
   913→                    data[i].clientOrderId = SignalUtils.generateOrderId(data[i].timestamp, symbol);
   914→                    this.signals.push({
   915→                        type: 'Short',
   916→                        timestamp: data[i].timestamp,

   918→                        tp: data[i].tp,

   929→        // Store processed data

   937→    /**
   938→     * محاسبه وضعیت سیگنال (در انتظار / باز شده / بسته شده)
   939→     * (کپی از داشبورد)
   940→     */
   941→    /**
   942→     * ✓ محاسبه وضعیت سیگنال (در انتظار / باز شده / باز نشده / منقضی شده)
   943→     *
   944→     * ✓ منطق انقضای سیگنال (مورد ۴):
   945→     *   اگر اختلاف زمان فعلی و زمان صدور سیگنال بیشتر از signalExpirationHours باشد
   946→     *   و پوزیشن هم‌جهتی برای آن نماد باز نشده باشد → وضعیت: «منقضی شده»
   947→     */
   948→    calculateSignalStatus(signal) {
   949→        // Handle case where currentSymbolHistory is not yet loaded
   950→        if (!this.currentSymbolHistory || !Array.isArray(this.currentSymbolHistory)) {
   951→            const isLatest = this.signals.indexOf(signal) === this.signals.length - 1;
   952→            // ✓ Check expiration even without history
   953→            if (isLatest && this.isSignalExpired(signal)) {
   954→                return { text: 'منقضی شده', color: 'text-orange-400' };
   955→            }
   956→            return {
   957→                text: isLatest ? 'در انتظار' : 'باز نشده',
   958→                color: isLatest ? 'text-yellow-400' : 'text-gray-400'
   959→            };
   960→        }
   961→        
   962→        // Filter OPEN positions with same direction as signal
   963→        // BUY_OPEN → Long, SELL_OPEN → Short
   964→        const signalDirection = signal.type === 'Long' ? 'BUY' : 'SELL';
   965→        const openPositions = this.currentSymbolHistory.filter(pos => 
   966→            pos.side && pos.side.includes('OPEN') && pos.side.includes(signalDirection)
   967→        );
   968→        
   969→        const currentIndex = this.signals.indexOf(signal);
   970→        const nextSignal = currentIndex < this.signals.length - 1 ? this.signals[currentIndex + 1] : null;
   971→        
   972→        const currentTime = signal.timestamp.getTime();
   973→        const nextTime = nextSignal ? nextSignal.timestamp.getTime() : Infinity;
   974→        
   975→        // Match: same-direction OPEN position in time range [T_i, T_{i+1})
   976→        const matchingPositions = openPositions.filter(pos => {
   977→            const posTime = pos.time.getTime();
   978→            return posTime >= currentTime && posTime < nextTime;
   979→        });
   980→        
   981→        if (matchingPositions.length === 0) {
   982→            // No matching position found
   983→            const isLatest = currentIndex === this.signals.length - 1;
   984→            
   985→            // ✓ Check if signal is expired (مورد ۴)
   986→            if (isLatest && this.isSignalExpired(signal)) {
   987→                return { text: 'منقضی شده', color: 'text-orange-400' };
   988→            }
   989→            
   990→            return {
   991→                text: isLatest ? 'در انتظار' : 'باز نشده',
   992→                color: isLatest ? 'text-yellow-400' : 'text-gray-400'
   993→            };
   994→        }
   995→        
   996→        const earliestPosition = matchingPositions.reduce((earliest, current) => {
   997→            return current.time.getTime() < earliest.time.getTime() ? current : earliest;
   998→        });
   999→        
  1000→        return {
  1001→            text: earliestPosition.time.toLocaleString('fa-IR', {
  1002→                year: 'numeric',
  1003→                month: '2-digit',
  1004→                day: '2-digit',
  1005→                hour: '2-digit',
  1006→                minute: '2-digit'
  1007→            }),
  1008→            color: 'text-green-400'
  1009→        };
  1010→    }
  1011→
  1012→    /**
  1013→     * ✓ بررسی انقضای سیگنال (مورد ۴)
  1014→     *
  1015→     * اگر اختلاف زمان فعلی و زمان صدور سیگنال بیشتر از signalExpirationHours باشد → true
  1016→     */
  1017→    isSignalExpired(signal) {
  1018→        if (!signal || !signal.timestamp) return false;
  1019→        const signalTime = signal.timestamp instanceof Date 
  1020→            ? signal.timestamp.getTime() 
  1021→            : new Date(signal.timestamp).getTime();
  1022→        const now = Date.now();
  1023→        const expirationMs = (this.settings.signalExpirationHours || 6) * 60 * 60 * 1000;
  1024→        return (now - signalTime) > expirationMs;
  1025→    }
