     1→/**
     2→ * Server-Side Cycle Engine
     3→ * =========================
     4→ *
     5→ * A faithful TypeScript port of the client-side `AutomationManager.runCycle()`
     6→ * logic (from public/js/automation-manager.js) so that a full automation cycle
     7→ * can run on the server — triggered either manually via /api/run-cycle or
     8→ * automatically via the Cloudflare Cron Worker (every 5 minutes).
     9→ *
    10→ * CRITICAL DESIGN RULES:
    11→ *  - The logic MUST exactly mirror the client-side runCycle() so that
    12→ *    behaviour is identical whether triggered from the browser button
    13→ *    or from the cron worker. Do not "improve" or refactor the logic.
    14→ *  - All settings are read from the permanent database (AutomationState),
    15→ *    NOT from localStorage. This is the key Cloudflare migration requirement.
    16→ *  - Indicator math (RSI/ATR/SMA, signal generation) is copied verbatim
    17→ *    from public/js/shared/signal-utils.js and analyzeMarketData().
    18→ *  - Risk controls (safe asset, price distance, max margin, positive margin)
    19→ *    use the same formulas and produce the same Bale notifications.
    20→ *
    21→ * Flow (mirrors runCycle() 14 steps):
    22→ *   1. selectNextSymbol()   — pick first ready symbol (respects tradeWaitTime)
    23→ *   2. setupForCycle()      — clear per-cycle state
    24→ *   3. fetchMarketData()    — GET /api/toobit-proxy (klines)
    25→ *   4. analyzeMarketData()  — generate signals (RSI/ATR/SMA/daily levels)
    26→ *   5. fetchPositionHistory — POST /api/history (mapped to consistent shape)
    27→ *   6. updateSelectedSignal — only latest signal with status "در انتظار"
    28→ *   7. fetchPrice()         — current price
    29→ *   8. closeOppositePositions
    30→ *   9. getSymbolMargin()    — existing same-direction margin
    31→ *  10. fetchBalance()       — total / free / positionMargin
    32→ *  11. calculateNewMargin()
    33→ *  12. Risk controls (safe asset, price distance, max margin, positive margin)
    34→ *  13. openPosition()       — POST /api/create-position
    35→ *  14. notifyOpenPosition() — POST /api/bale-send
    36→ *
    37→ * After success or skip, the symbol's status/lastCycleTime/errorCount are
    38→ * persisted back to the database.
    39→ */

   203→function calculateRSI(data: Candle[], period = 14): (number | null)[] {
   204→  let avgGain = 0
   205→  let avgLoss = 0
   206→  const rsiValues: (number | null)[] = []
   207→
   208→  if (data.length < period) {
   209→    return new Array(data.length).fill(null)
   210→  }
   211→
   212→  for (let i = 1; i < data.length; i++) {
   213→    const change = data[i].close - data[i - 1].close
   214→    const gain = Math.max(change, 0)
   215→    const loss = Math.max(-change, 0)
   216→
   217→    if (i <= period) {
   218→      avgGain += gain
   219→      avgLoss += loss
   220→      if (i === period) {
   221→        avgGain /= period
   222→        avgLoss /= period
   223→      }
   224→      rsiValues.push(null)
   225→    } else {
   226→      avgGain = (avgGain * (period - 1) + gain) / period
   227→      avgLoss = (avgLoss * (period - 1) + loss) / period
   228→      const rs = avgLoss === 0 ? 100 : avgGain / avgLoss
   229→      rsiValues.push(100 - 100 / (1 + rs))
   230→    }
   231→  }
   232→
   233→  return rsiValues
   234→}

   236→function calculateATR(data: Candle[], period = 14): number[] {
   237→  const atrValues: number[] = []
   238→
   239→  const trueRanges = data.map((candle, i) => {
   240→    if (i === 0) return candle.high - candle.low
   241→    const prevClose = data[i - 1].close
   242→    return Math.max(
   243→      candle.high - candle.low,
   244→      Math.abs(candle.high - prevClose),
   245→      Math.abs(candle.low - prevClose)
   246→    )
   247→  })
   248→
   249→  atrValues[0] = trueRanges[0]
   250→  for (let i = 1; i < data.length; i++) {
   251→    atrValues[i] = ((period - 1) * atrValues[i - 1] + trueRanges[i]) / period
   252→  }
   253→
   254→  return atrValues
   255→}
   256→
   257→function calculateSMA(data: number[], period: number): (number | null)[] {
   258→  const smaValues: (number | null)[] = []
   259→  for (let i = 0; i < data.length; i++) {
   260→    if (i < period - 1) {
   261→      smaValues.push(null)
   262→    } else {
   263→      let sum = 0
   264→      for (let j = i - period + 1; j <= i; j++) {
   265→        sum += data[j]
   266→      }
   267→      smaValues.push(sum / period)
   268→    }
   269→  }
   270→  return smaValues
   271→}

   273→function generateOrderId(timestamp: number | Date, symbol: string): string {
   274→  const dt = new Date(timestamp)
   275→  const year = dt.getFullYear().toString().padStart(4, '0')
   276→  const month = (dt.getMonth() + 1).toString().padStart(2, '0')
   277→  const day = dt.getDate().toString().padStart(2, '0')
   278→  const hour = dt.getHours().toString().padStart(2, '0')
   279→  const minute = dt.getMinutes().toString().padStart(2, '0')
   280→  const second = dt.getSeconds().toString().padStart(2, '0')
   281→  return `${symbol}_${year}${month}${day}_${hour}${minute}${second}`
   282→}

   285→// Jalali date conversion — verbatim port of toJalali() in automation-manager.js

   288→function toJalali(date?: Date | number): string {
   289→  if (!date) date = new Date()
   290→  if (typeof date === 'number') date = new Date(date)
   291→
   292→  const tehranOffset = 3.5 * 60 // minutes
   293→  const utcTime = date.getTime() + date.getTimezoneOffset() * 60000
   294→  const tehranTime = new Date(utcTime + tehranOffset * 60000)
   295→
   296→  const gy = tehranTime.getFullYear()
   297→  const gm = tehranTime.getMonth() + 1
   298→  const gd = tehranTime.getDate()
   299→
   300→  const g_d_m = [0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334]
   301→  let jy: number, jm: number, jd: number, gy2: number, days: number
   302→
   303→  gy2 = gm > 2 ? gy + 1 : gy
   304→  days =
   305→    355666 +
   306→    365 * gy +
   307→    Math.floor((gy2 + 3) / 4) -
   308→    Math.floor((gy2 + 99) / 100) +
   309→    Math.floor((gy2 + 399) / 400) +
   310→    gd +
   311→    g_d_m[gm - 1]
   312→  jy = -1595 + 33 * Math.floor(days / 12053)
   313→  days %= 12053
   314→  jy += 4 * Math.floor(days / 1461)
   315→  days %= 1461
   316→  if (days > 365) {
   317→    jy += Math.floor((days - 1) / 365)
   318→    days = (days - 1) % 365
   319→  }
   320→  if (days < 186) {
   321→    jm = 1 + Math.floor(days / 31)
   322→    jd = 1 + (days % 31)
   323→  } else {
   324→    jm = 7 + Math.floor((days - 186) / 30)
   325→    jd = 1 + ((days - 186) % 30)
   326→  }
   327→
   328→  const hours = String(tehranTime.getHours()).padStart(2, '0')
   329→  const minutes = String(tehranTime.getMinutes()).padStart(2, '0')
   330→  const seconds = String(tehranTime.getSeconds()).padStart(2, '0')
   331→
   332→  return `${jy}/${String(jm).padStart(2, '0')}/${String(jd).padStart(2, '0')} - ${hours}:${minutes}:${seconds}`
   333→}

   339→/**
   340→ * Execute one full automation cycle on the server.
   341→ *
   342→ * @param baseUrl - the base URL of the running app (for internal API calls).
   343→ *                  In local dev this is http://localhost:3000.
   344→ *                  In Cloudflare this is the deployed Pages URL.
   345→ * @param options - optional overrides (used by tests / manual triggers)
   346→ */
   347→export async function runCycle(
   348→  baseUrl: string,
   349→  options: {
   350→    symbolOverride?: string | null
   351→    triggerSource?: 'manual' | 'cron'
   352→  } = {}
   353→): Promise<CycleResult> {
   354→  const startTime = Date.now()
   355→  const logs: CycleLogEntry[] = []
   356→  // ✓ Batch logging: collect all logs in memory during the cycle, then flush
   357→  //   them to KV in a SINGLE write at the end. This avoids the race condition
   358→  //   where concurrent cron cycles overwrite each other's logs (each log was
   359→  //   previously doing a full read-modify-write of the entire logs array).
   360→  const pendingDbLogs: Array<{
   361→    message: string
   362→    type: string
   363→    symbol: string
   364→    source: string
   365→    timestamp: string
   366→  }> = []
   367→
   368→  const log = (message: string, type: CycleLogEntry['type'] = 'info', symbol?: string) => {
   369→    const entry: CycleLogEntry = { message, type, symbol, timestamp: new Date().toISOString() }
   370→    logs.push(entry)
   371→    // Buffer for batch write — do NOT write to KV on each log call
   372→    pendingDbLogs.push({
   373→      message,
   374→      type,
   375→      symbol: symbol || '',
   376→      source: options.triggerSource || 'server',
   377→      timestamp: entry.timestamp
   378→    })
   379→    console.log(`[CycleEngine${options.triggerSource ? `:${options.triggerSource}` : ''}] ${type.toUpperCase()} ${symbol ? `[${symbol}] ` : ''}${message}`)
   380→  }
   381→
   382→  // ✓ Flush all buffered logs to KV.
   383→  //   Uses ONE write strategy (Task 11 — KV write reduction):
   384→  //     1. `lastCycleLogs` — stores THIS cycle's complete logs (single KV put).
   385→  //   The shared historical `logs` array is NO LONGER updated per-cycle.
   386→  //   Instead, /api/run-cycle does ONE batch append of all cycle logs after
   387→  //   all cycles in a single cron call complete (via AutomationLogs.appendBatch).
   388→  // ✓ Transaction Buffer (پیشنهاد ۱): flushLogs دیگر به KV نمی‌نویسد
   389→  //   logs در CycleResult.logs برگردانده می‌شوند و /api/run-cycle در پایان
   390→  //   یک‌بار آنها را به lastCycleLogs و shared logs می‌نویسد
   391→  const flushLogs = async () => {
   392→    // No-op: logs are already in the `logs` array and will be returned in CycleResult
   393→    // The caller (/api/run-cycle) is responsible for writing them to KV
   394→  }
   395→
   396→  try {
   397→    // -------------------------------------------------------------------
   398→    // Load settings + symbols from the permanent database (async — KV)
   399→    // -------------------------------------------------------------------
   400→    const dbAutomationSettings = await AutomationState.get('automation_settings')
   401→    const automationSettings: AutomationSettings = {
   402→      ...getDefaultAutomationSettings(),
   403→      ...(typeof dbAutomationSettings === 'string'
   404→        ? JSON.parse(dbAutomationSettings)
   405→        : (dbAutomationSettings as Record<string, unknown>) || {})
   406→    }
   407→
   408→    const dbMarketSettings = await AutomationState.get('marketSignalSettings')
   409→    const marketSettings: MarketSignalSettings =
   410→      typeof dbMarketSettings === 'string'
   411→        ? JSON.parse(dbMarketSettings)
   412→        : (dbMarketSettings as MarketSignalSettings) || {}
   413→
   414→    const dbSymbols = await AutomationState.get('automation_symbols')
   415→    const symbols: SymbolEntry[] = (() => {
   416→      const parsed =
   417→        typeof dbSymbols === 'string' ? JSON.parse(dbSymbols) : (dbSymbols as SymbolEntry[]) || []
   418→      return Array.isArray(parsed) ? parsed : []
   419→    })()
   420→
   421→    if (symbols.length === 0) {
   422→      log('هیچ نمادی در لیست اتوماسیون وجود ندارد', 'warning')
   423→      return {
   424→        success: false,
   425→        symbol: null,
   426→        action: 'no_ready_symbol',
   427→        logs,
   428→        errorCount: 0,
   429→        duration: Date.now() - startTime,
   430→        symbols
   431→      }
   432→    }
   433→
   434→    // -------------------------------------------------------------------
   435→    // Step 1: selectNextSymbol()
   436→    // -------------------------------------------------------------------
   437→    const now = Date.now()
   438→    const waitTimeMs = automationSettings.tradeWaitTime * 60 * 1000
   439→    const readySymbols = symbols.filter((s) => {
   440→      if (!s.lastCycleTime) return true
   441→      return now - s.lastCycleTime >= waitTimeMs
   442→    })
   443→
   444→    let symbol: SymbolEntry | null = null
   445→    if (options.symbolOverride) {
   446→      symbol = symbols.find((s) => s.name === options.symbolOverride) || null
   447→    } else {
   448→      symbol = readySymbols.length > 0 ? readySymbols[0] : null
   449→    }
   450→
   451→    if (!symbol) {
   452→      log('هیچ نمادی برای معامله آماده نیست', 'warning')
   453→      return {
   454→        success: false,
   455→        symbol: null,
   456→        action: 'no_ready_symbol',
   457→        logs,
   458→        errorCount: 0,
   459→        duration: Date.now() - startTime,
   460→        symbols
   461→      }
   462→    }
   463→
   464→    log(`شروع چرخه برای نماد ${symbol.name}`, 'info', symbol.name)
   465→
   466→    // -------------------------------------------------------------------
   467→    // Helper to persist symbol state back to DB
   468→    // ✓ IMPORTANT: now async (await) — caller MUST await to ensure KV consistency
   469→    //
   470→    // ✓ Task 11 (KV write reduction): We have TWO helpers now:
   471→    //   - `updateSymbolState` (async) — does the actual KV write. Used ONLY at
   472→    //     the START of a cycle to set `status: 'running'` + `lastCycleTime`
   473→    //     (one write per cycle, persisted immediately to prevent the same
   474→    //     symbol being picked again by the next cycle).
   475→    //   - `updateSymbolStateInMemory` (sync, no KV write) — used at the END of
   476→    //     every cycle (success / no_signal / risk_blocked) to flip `status`
   477→    //     back to 'waiting'. The `lastCycleTime` was already persisted at the
   478→    //     START, so the next cycle's selectNextSymbol will skip this symbol.
   479→    //     The status change is cosmetic (UI reads it) — not worth a KV write.
   480→    //
   481→    //     The ONE exception is the catch block (error path): it does its own
   482→    //     read-modify-write of `automation_symbols` to bump `errorCount`,
   483→    //     which MUST be persisted (allowedErrors tracking). That write is
   484→    //     kept as-is — only the redundant end-of-cycle 'waiting' writes are
   485→    //     removed.
   486→    // -------------------------------------------------------------------
   487→    // ✓ Transaction Buffer (پیشنهاد ۱+۶): تمام state updates در حافظه انجام می‌شود
   488→    //   KV write فقط در پایان /api/run-cycle انجام می‌شود (نه در هر چرخه)
   489→    //   این KV writes را از ~۵ به ~۱-۲ در هر چرخه کاهش می‌دهد
   490→    const updateSymbolState = async (updates: Partial<SymbolEntry>) => {
   491→      // ✓ In-memory only — NO KV write during cycle
   492→      const idx = symbols.findIndex((s) => s.id === symbol!.id)
   493→      if (idx >= 0) {
   494→        symbols[idx] = { ...symbols[idx], ...updates }
   495→      }
   496→    }
   497→
   498→    const updateSymbolStateInMemory = (updates: Partial<SymbolEntry>) => {
   499→      const idx = symbols.findIndex((s) => s.id === symbol!.id)
   500→      if (idx >= 0) {
   501→        symbols[idx] = { ...symbols[idx], ...updates }
   502→      }
   503→    }
   504→
   505→    // ✓ FIX (مشکل ۱): Set lastCycleTime IMMEDIATELY in-memory after symbol selection
   506→    //   KV write deferred to end of /api/run-cycle (Transaction Buffer)
   507→    updateSymbolStateInMemory({ lastCycleTime: Date.now() })
   508→
   509→    // -------------------------------------------------------------------
   510→    // Step 2-3: fetchMarketData()
   511→    // -------------------------------------------------------------------
   512→    const interval = marketSettings.interval || '1h'
   513→    const limit = marketSettings.limit || 1000
   514→    const fullSymbol = `${symbol.name}-SWAP-USDT`
   515→    const proxyUrl = `${baseUrl}/api/toobit-proxy?symbol=${encodeURIComponent(fullSymbol)}&interval=${encodeURIComponent(interval)}&limit=${encodeURIComponent(limit)}`
   516→
   517→    log(`${symbol.name}: دریافت داده‌های بازار...`, 'info', symbol.name)
   518→    const marketResponse = await fetch(proxyUrl)
   519→    if (!marketResponse.ok) {
   520→      throw new Error(`Failed to fetch market data: HTTP ${marketResponse.status}`)
   521→    }
   522→    const rawData = await marketResponse.json()
   523→    if (!Array.isArray(rawData) || rawData.length === 0 || !Array.isArray(rawData[0])) {
   524→      throw new Error('داده‌های بازار معتبر نیست')
   525→    }
   526→    log(`${symbol.name}: ${rawData.length} کندل دریافت شد`, 'success', symbol.name)
   527→
   528→    // -------------------------------------------------------------------
   529→    // Step 4: analyzeMarketData() — signal generation (verbatim port)
   530→    // -------------------------------------------------------------------
   531→    const N = rawData.length
   532→    const data: Candle[] = rawData.map((c: unknown[]) => ({
   533→      timestamp: new Date(c[0] as number),
   534→      open: parseFloat(c[1] as string),
   535→      high: parseFloat(c[2] as string),
   536→      low: parseFloat(c[3] as string),
   537→      close: parseFloat(c[4] as string),
   538→      amount: parseFloat(c[5] as string)
   539→    }))
   540→
   541→    // Calculate daily data (Tehran offset +3:30)
   542→    const offset = 3.5 * 3600 * 1000

  1447→        // fall through to Method 2
