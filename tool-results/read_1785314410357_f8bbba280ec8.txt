   419→            }

   445→
   446→            async dbGet(key) {
   447→                try {
   448→                    const response = await fetch(`/api/db/automation-state?key=${encodeURIComponent(key)}`);
   449→                    if (!response.ok) return null;
   450→                    const data = await response.json();
   451→                    // API returns { success: true, data: value }
   452→                    if (data.success && data.data !== undefined && data.data !== null) {
   453→                        return data.data;
   454→                    }
   455→                    return data.value !== undefined ? data.value : null;
   456→                } catch (e) { return null; }
   457→            }
   458→
   459→            async dbSet(key, value) {
   460→                try {
   461→                    await fetch('/api/db/automation-state', {
   462→                        method: 'POST',
   463→                        headers: { 'Content-Type': 'application/json' },
   464→                        body: JSON.stringify({ key, value })
   465→                    });
   466→                } catch (e) { /* ignore */ }
   467→            }
   468→
   469→            async dbDelete(key) {
   470→                try {
   471→                    await fetch(`/api/db/automation-state?key=${encodeURIComponent(key)}`, { method: 'DELETE' });
   472→                } catch (e) { /* ignore */ }
   473→            }
   474→
   475→            async loadApiKeysFromDB() {
   476→                try {

   478→                    if (dbKeys) {
   479→                        const keys = typeof dbKeys === 'string' ? JSON.parse(dbKeys) : dbKeys;
   480→                        if (keys.apiKey) document.getElementById('api-key').value = keys.apiKey;
   481→                        if (keys.secretKey) document.getElementById('secret-key').value = keys.secretKey;

   487→            /**
   488→             * ✓ Cloudflare-ready: Save the full marketSignalSettings object to the database.
   489→             * This ensures ALL settings (indicators, symbol, API keys, etc.) are persisted
   490→             * server-side and available to the cron-triggered run-cycle.
   491→             */
   492→            async saveSettingsToDB(settings) {
   493→                try {
   494→                    // Save the complete settings object under 'marketSignalSettings' key
   495→                    await this.dbSet('marketSignalSettings', settings);
   496→                    // Also keep the apiKeys entry for backwards compatibility
   497→                    await this.dbSet('settings_apiKeys', {
   498→                        apiKey: settings.apiKey || '',
   499→                        secretKey: settings.secretKey || '',
   500→                        baseUrl: settings.baseUrl || 'https://api.toobit.com'
   501→                    });
   502→                    console.log('✓ Settings saved to database (permanent storage)');
   503→                } catch (e) {
   504→                    console.warn('Could not save settings to database:', e);
   505→                }
   506→            }

   508→            /**
   509→             * ✓ Cloudflare-ready: Load full settings from the database on page load.
   510→             * DB takes priority over localStorage. Falls back gracefully if DB is unavailable.
   511→             */
   512→            async loadSettingsFromDB() {
   513→                try {
   514→                    const dbSettings = await this.dbGet('marketSignalSettings');
   515→                    if (dbSettings) {
   516→                        const settings = typeof dbSettings === 'string' ? JSON.parse(dbSettings) : dbSettings;
   517→                        // Merge DB settings over localStorage (DB wins)
   518→                        const local = JSON.parse(localStorage.getItem('marketSignalSettings') || '{}');
   519→                        const merged = { ...this.getDefaultSettings(), ...local, ...settings };
   520→                        localStorage.setItem('marketSignalSettings', JSON.stringify(merged));
   521→                        return merged;
   522→                    }
   523→                } catch (e) { /* ignore */ }
   524→                return null;
   525→            }

   527→            loadSettings() {
   528→                const saved = localStorage.getItem('marketSignalSettings');
   529→                const settings = saved ? JSON.parse(saved) : this.getDefaultSettings();
   530→                
   531→                // Apply settings to form
   532→                Object.keys(settings).forEach(key => {
   533→                    const element = document.getElementById(key.replace(/([A-Z])/g, '-$1').toLowerCase());
   534→                    if (element) {
   535→                        if (settings[key] === null || settings[key] === '') {
   536→                            element.value = '';
   537→                        } else {
   538→                            element.value = settings[key];
   539→                        }
   540→                    }
   541→                });
   542→
   543→                // Load API keys from DB (takes priority)
   544→                this.loadApiKeysFromDB();
   545→            }

   547→            /**
   548→             * ✓ Cloudflare-ready: Async load full settings from database.
   549→             * DB takes priority over localStorage. Re-populates the form if DB has newer data.
   550→             */
   551→            async loadSettingsAsync() {
   552→                const dbSettings = await this.loadSettingsFromDB();
   553→                if (dbSettings) {
   554→                    // Re-apply merged settings to the form
   555→                    Object.keys(dbSettings).forEach(key => {
   556→                        const element = document.getElementById(key.replace(/([A-Z])/g, '-$1').toLowerCase());
   557→                        if (element) {
   558→                            if (dbSettings[key] === null || dbSettings[key] === '') {
   559→                                element.value = '';
   560→                            } else {
   561→                                element.value = dbSettings[key];
   562→                            }
   563→                        }
   564→                    });
   565→                    console.log('✓ Settings loaded from database (permanent storage)');
   566→                }
   567→            }

   569→            saveSettings() {
   570→                const settings = {};
   571→                const inputs = document.querySelectorAll('input, select');
   572→                
   573→                inputs.forEach(input => {
   574→                    const key = input.id.replace(/-([a-z])/g, (g) => g[1].toUpperCase());
   575→                    if (input.type === 'number') {
   576→                        settings[key] = input.value === '' ? null : parseFloat(input.value);
   577→                    } else {
   578→                        // تبدیل نماد به حروف بزرگ قبل از ذخیره
   579→                        if (key === 'symbolName') {
   580→                            settings[key] = input.value.toUpperCase();
   581→                        } else {
   582→                            settings[key] = input.value;
   583→                        }
   584→                    }
   585→                });
   586→                
   587→                localStorage.setItem('marketSignalSettings', JSON.stringify(settings));
   588→
   589→                // ✓ Cloudflare-ready: Save ALL settings to the permanent database (KV in Cloudflare)
   590→                // This makes settings available to server-side operations (cron-triggered run-cycle)
   591→                // and persists across devices/browsers.
   592→                this.saveSettingsToDB(settings);
   593→
   594→                // Show success message
   595→                this.showNotification('تنظیمات با موفقیت ذخیره شد!', 'success');
   596→            }

   600→                    localStorage.removeItem('marketSignalSettings');
   601→                    // ✓ Also clear from permanent database

   605→                    this.showNotification('تنظیمات به حالت پیش‌فرض بازنشانی شد.', 'info');
   606→                }
   607→            }
